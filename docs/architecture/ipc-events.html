close_document doc_id, discard_changes CloseResult Closes document, optionally discards changes
reload_file path Document Reloads file from disk
create_document — Document Creates new untitled document
check_file_modified path FileModifiedResult Checks if file was modified externally

Events (emit/listen)

🎯

menu:action

Emitted when user clicks a native menu item.

Payload
{
  action: string,    // "file.new", "file.open", etc.
  value?: string     // Optional parameter
}
Actions:
file.new
file.open
file.save
file.save_as
file.close
file.export_html
set-view-mode
cycle-view-mode
view.toggle_sidebar
format.highlight
theme.light
theme.dark
theme.system
settings.open
Source: menu.rs
📂

file:open

Emitted when files are opened via Finder (file association) or drag-drop.

Payload
string[]  // Array of file paths
Source: launch.rs, main.rs
👁

file:changed

Emitted when an open file is modified externally.

Payload
{
  path: string,
  changeType: "modified" | "deleted"
}
Source: services/file_watcher.rs
🎨

theme:changed

Emitted when system theme changes (when in "system" mode).

Payload
"light" | "dark"
Source: main.rs

Usage Examples

Calling a Command (Frontend)

import { invoke } from '@tauri-apps/api/core';

// Open a file
const result = await invoke<OpenFileResult>(
  'open_file',
  {
    path: '/path/to/file.md',
    encoding: null,
    actionId: 'action-123'
  }
);

// Save a file
await invoke('save_file', {
  documentId: doc.id,
  content: doc.content,
  path: doc.path,
  actionId: 'action-456'
});

Listening to Events (Frontend)

import { listen } from '@tauri-apps/api/event';

// Listen to menu actions
const unlisten = await listen<MenuAction>(
  'menu:action',
  (event) => {
    const { action, value } = event.payload;
    switch (action) {
      case 'file.new':
        createNewDocument();
        break;
      case 'file.save':
        saveActiveDocument();
        break;
    }
  }
);

// Cleanup on unmount
unlisten();

Defining a Command (Backend)

use tauri::command;

#[command]
pub fn open_file(
    path: String,
    encoding: Option<String>,
    action_id: String,
) -> Result<OpenFileResult, String> {
    // Log start
    logger::log_event(LogEntryPayload {
        level: LogLevel::Info,
        event: "file.open".to_string(),
        status: LogStatus::Start,
        action_id: Some(action_id.clone()),
        ..Default::default()
    });

    // Read file
    let content = std::fs::read_to_string(&path)
        .map_err(|e| e.to_string())?;

    // Return result
    Ok(OpenFileResult { ... })
}

Emitting an Event (Backend)

use tauri::Manager;

// In menu handler
fn handle_menu_event(
    app: &AppHandle,
    event: MenuEvent
) {
    let action = match event.id.as_str() {
        "new" => "file.new",
        "open" => "file.open",
        "save" => "file.save",
        _ => return,
    };

    app.emit("menu:action", MenuActionPayload {
        action: action.to_string(),
        value: None,
    }).ok();
}

// In file watcher
app.emit("file:changed", FileChangedPayload {
    path: path.to_string(),
    change_type: "modified".to_string(),
}).ok();

IPC Data Types

TypeScript (Frontend)

interface OpenFileResult {
  document: Document;
  needsEncodingSelection: boolean;
  detectedEncoding: string;
}

interface SaveResult {
  success: boolean;
  path: string;
}

interface Document {
  id: string;
  path: string | null;
  content: string;
  encoding: string;
  isModified: boolean;
  createdAt: string;
  modifiedAt: string;
}

interface MenuAction {
  action: string;
  value?: string;
}

Rust (Backend)

#[derive(Serialize)]
pub struct OpenFileResult {
    pub document: Document,
    pub needs_encoding_selection: bool,
    pub detected_encoding: String,
}

#[derive(Serialize)]
pub struct SaveResult {
    pub success: bool,
    pub path: String,
}

#[derive(Serialize, Deserialize)]
pub struct Document {
    pub id: String,
    pub path: Option<PathBuf>,
    pub content: String,
    pub encoding: String,
    pub is_modified: bool,
    pub created_at: DateTime<Utc>,
    pub modified_at: DateTime<Utc>,
}
