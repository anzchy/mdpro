Plugin Registry

Core Functions

register(plugin)

Registers a plugin with the registry

initialize(mode, context)

Initializes all plugins for editor mode

getCodemirrorExtensions()

Returns all CodeMirror extensions

getMilkdownPlugins()

Returns all Milkdown plugins

emitEvent(event, data)

Broadcasts events to all plugins

destroy()

Cleans up all plugins

Usage

// In Editor.tsx
import { pluginRegistry } from '@/plugins';

// Initialize for mode
pluginRegistry.initialize('codemirror', {
  document: activeDoc,
  editorView: viewRef.current,
  theme: resolvedTheme,
});

// Get extensions for CodeMirror
const extensions = pluginRegistry
  .getCodemirrorExtensions();

// Emit editor event
pluginRegistry.emitEvent('doc:change', {
  content: newContent
});

Plugin Interface

interface EditorPlugin {
  // Unique plugin identifier
  name: string;

  // Which editor modes this plugin supports
  supportedModes: ('codemirror' | 'milkdown')[];

  // Default configuration
  defaultConfig: {
    enabled: boolean;
    priority: number;  // Lower = loaded first
    [key: string]: any;
  };

  // CodeMirror extension factory (optional)
  codemirrorExtension?: (
    config: PluginConfig,
    context: PluginContext
  ) => Extension[];

  // Milkdown plugin factory (optional)
  milkdownPlugin?: (
    config: PluginConfig,
    context: PluginContext
  ) => unknown[];

  // Lifecycle hooks
  init?: (context: PluginContext, config: PluginConfig) => void;
  destroy?: (context: PluginContext, config: PluginConfig) => void;

  // Event handler
  onEvent?: (event: string, data: any) => void;
}

Creating a Plugin

CodeMirror Plugin Example

import { EditorPlugin } from '../core/pluginTypes';
import { Decoration } from '@codemirror/view';

export const myPlugin: EditorPlugin = {
  name: 'my-plugin',
  supportedModes: ['codemirror'],
  defaultConfig: {
    enabled: true,
    priority: 50,
    customOption: 'value',
  },

  codemirrorExtension(config, context) {
    return [
      // Return CodeMirror extensions
      ViewPlugin.fromClass(class {
        decorations = Decoration.none;

        update(update) {
          // Handle updates
        }
      }, {
        decorations: v => v.decorations
      })
    ];
  },

  init(context, config) {
    console.log('Plugin initialized');
  },

  destroy(context, config) {
    console.log('Plugin destroyed');
  },
};

Milkdown Plugin Example

import { EditorPlugin } from '../core/pluginTypes';
import { $prose } from '@milkdown/kit/utils';

export const myMilkdownPlugin: EditorPlugin = {
  name: 'my-milkdown-plugin',
  supportedModes: ['milkdown'],
  defaultConfig: {
    enabled: true,
    priority: 50,
  },

  milkdownPlugin(config, context) {
    return [
      $prose((ctx) => {
        return new Plugin({
          key: new PluginKey('my-plugin'),

          props: {
            handleKeyDown(view, event) {
              // Custom key handling
              return false;
            },
          },
        });
      }),
    ];
  },

  onEvent(event, data) {
    if (event === 'doc:change') {
      // React to document changes
    }
  },
};

Register in index.ts

// src/plugins/index.ts
import { pluginRegistry } from './core/pluginRegistry';
import { myPlugin } from './codemirror/myPlugin';
import { myMilkdownPlugin } from './milkdown/myMilkdownPlugin';

// Register all plugins
pluginRegistry.register(myPlugin);
pluginRegistry.register(myMilkdownPlugin);

export { pluginRegistry };

Runtime Configuration

// Get current config
const config = pluginRegistry.getConfig('focus-mode');

// Update config
pluginRegistry.setConfig('focus-mode', {
  enabled: true,
  opacity: 0.5,
});

// Enable/disable plugin
pluginRegistry.enable('focus-mode');
pluginRegistry.disable('typewriter-mode');

// Check if enabled
const isEnabled = pluginRegistry.isEnabled('focus-mode');

Plugin Events

Event Payload Description
doc:change { content: string } Document content changed
selection:change { from: number, to: number } Selection changed
focus { focused: boolean } Editor focus state changed
theme:change { theme: "light" | "dark" } Theme changed
mode:change { mode: ViewMode } Editor mode changed
